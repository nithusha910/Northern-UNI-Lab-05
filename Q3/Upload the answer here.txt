import java.util.Scanner;

public class IT26102586Lab5Q3 {
    public static void main(String[] args) {
        // CONSTANT da
        final double ROOM_CHARGE_PER_DAY = 48000.0;
        final double DISCOUNT_RATE_1 = 10.0;
        final double DISCOUNT_RATE_2 = 20.0;

        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter Start Date (1-31): ");
        int startDate = scanner.nextInt();

        System.out.print("Enter End Date (1-31): ");
        int endDate = scanner.nextInt();

        // Validation 1
        if (startDate < 1 || startDate > 31 || endDate < 1 || endDate > 31) {
            System.out.println("Error: Days must be between 1 and 31");
            return;
        }

        // Validation 2
        if (startDate >= endDate) {
            System.out.println("Error: Start Date must be less than End Date");
            return;
        }

        // Calculation
        int noOfDays = endDate - startDate;
        double totalCharge = noOfDays * ROOM_CHARGE_PER_DAY;
        double discount = 0;

        if (noOfDays >= 3 && noOfDays <= 4) {
            discount = totalCharge * (DISCOUNT_RATE_1 / 100);
        } else if (noOfDays >= 5) {
            discount = totalCharge * (DISCOUNT_RATE_2 / 100);
        }

        double amountToBePaid = totalCharge - discount;

        // Output
        System.out.println("\nRoom Charge Per Day: Rs. " + ROOM_CHARGE_PER_DAY + "/=");
        System.out.println("Number of Days Reserved: " + noOfDays);
        System.out.println("Total Amount to be Paid: " + amountToBePaid);
    }
}
